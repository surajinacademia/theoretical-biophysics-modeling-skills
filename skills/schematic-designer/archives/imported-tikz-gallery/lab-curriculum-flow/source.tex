\documentclass[tikz,border=8pt]{standalone}
\usepackage{scientific-neutral}
\usetikzlibrary{arrows.meta,backgrounds,fit,positioning}

\tikzset{
  lab/.style={
    draw=saBlue!75!black,
    fill=saBlue!10,
    rounded corners=3pt,
    minimum width=3.25cm,
    minimum height=0.82cm,
    align=center,
    line width=0.7pt
  },
  stage/.style={
    draw=saLightGray!90!black,
    fill=saPanelGray,
    rounded corners=5pt,
    inner sep=7pt,
    line width=0.55pt
  },
  prerequisite/.style={
    -{Stealth[length=2.2mm]},
    draw=saInk,
    line width=0.75pt
  },
  crosslink/.style={
    -{Stealth[length=2.0mm]},
    draw=saViolet,
    dashed,
    line width=0.7pt
  }
}

\begin{document}
\begin{tikzpicture}[node distance=8mm and 14mm]
  \node[lab] (signals) {Signals and systems};
  \node[lab,below=of signals] (fourier) {Fourier methods};
  \node[lab,below left=of fourier] (rf) {RF circuits};
  \node[lab,below right=of fourier] (sampling) {Sampling and data};
  \node[lab,below=13mm of rf] (modulation) {Modulation};
  \node[lab,below=13mm of sampling] (coding) {Digital coding};
  \node[lab,below=16mm of $(modulation)!0.5!(coding)$] (integration)
    {Integrated transceiver};
  \node[lab,below=of integration,fill=saGreen!12,draw=saGreen!75!black]
    (capstone) {Capstone measurement};

  \draw[prerequisite] (signals) -- (fourier);
  \draw[prerequisite] (fourier) -- (rf);
  \draw[prerequisite] (fourier) -- (sampling);
  \draw[prerequisite] (rf) -- (modulation);
  \draw[prerequisite] (sampling) -- (coding);
  \draw[prerequisite] (modulation) -- (integration);
  \draw[prerequisite] (coding) -- (integration);
  \draw[prerequisite] (integration) -- (capstone);
  \draw[crosslink] (rf.east) to[bend left=18] (coding.west);
  \draw[crosslink] (sampling.west) to[bend right=18] (modulation.east);

  \begin{scope}[on background layer]
    \node[stage,fit=(signals)(fourier),label={[sa/font small,text=saGray]north west:Foundations}] {};
    \node[stage,fit=(rf)(sampling)(modulation)(coding),
      label={[sa/font small,text=saGray]north west:Parallel laboratory streams}] {};
    \node[stage,fit=(integration)(capstone),
      label={[sa/font small,text=saGray]north west:Integration}] {};
  \end{scope}

  \node[anchor=south,sa/font title,text=saInk] at ([yshift=7mm]signals.north)
    {Laboratory curriculum dependencies};
  \node[anchor=north,sa/font tiny,text=saGray,align=center]
    at ([yshift=-6mm]capstone.south)
    {solid: prerequisite \quad dashed: cross-stream dependency};
\end{tikzpicture}
\end{document}
