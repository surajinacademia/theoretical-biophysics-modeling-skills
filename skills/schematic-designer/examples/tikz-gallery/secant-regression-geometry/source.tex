\documentclass[tikz,border=8pt]{standalone}
\usepackage{scientific-neutral}
\usetikzlibrary{arrows.meta,calc}

\begin{document}
\begin{tikzpicture}[x=1.15cm,y=1.15cm]
  \pgfmathsetmacro{\xP}{1.35}
  \pgfmathsetmacro{\xQ}{4.85}
  \pgfmathsetmacro{\yP}{0.16*(\xP-2.65)^2+0.90}
  \pgfmathsetmacro{\yQ}{0.16*(\xQ-2.65)^2+0.90}

  \coordinate (O) at (0,0);
  \coordinate (P) at (\xP,\yP);
  \coordinate (Q) at (\xQ,\yQ);
  \coordinate (Px) at (\xP,0);
  \coordinate (Qx) at (\xQ,0);
  \coordinate (R) at (\xQ,\yP);

  \draw[-{Stealth[length=2.3mm]},draw=saInk,line width=0.8pt]
    (-0.2,0) -- (6.35,0) node[below] {$x$};
  \draw[-{Stealth[length=2.3mm]},draw=saInk,line width=0.8pt]
    (0,-0.2) -- (0,3.25) node[left] {$f(x)$};

  \draw[draw=saRed,line width=1.1pt]
    plot[domain=0.15:6.05,samples=100,smooth]
      (\x,{0.16*(\x-2.65)^2+0.90});
  \node[anchor=west,sa/font small,text=saRed] at (5.75,2.65) {$f$};

  \draw[draw=saBlue,line width=1.0pt]
    ($(P)!-0.22!(Q)$) -- ($(P)!1.22!(Q)$)
    node[pos=0.92,above,sloped,sa/font small] {secant};

  \draw[draw=saGray,dashed] (P) -- (Px);
  \draw[draw=saGray,dashed] (Q) -- (Qx);
  \draw[draw=saGray,dashed] (P) -- (R) -- (Q);

  \fill[white,draw=saInk,line width=0.8pt] (P) circle (1.7pt)
    node[above left,sa/font small] {$P=(x_0,f(x_0))$};
  \fill[white,draw=saInk,line width=0.8pt] (Q) circle (1.7pt)
    node[above right,sa/font small] {$Q=(x_0+\Delta x,f(x_0+\Delta x))$};

  \draw[{Stealth[length=1.8mm]}-{Stealth[length=1.8mm]},draw=saGreen!80!black]
    ([yshift=-3mm]Px) -- node[below,sa/font small] {$\Delta x$} ([yshift=-3mm]Qx);
  \draw[{Stealth[length=1.8mm]}-{Stealth[length=1.8mm]},draw=saAmber!90!black]
    ([xshift=3mm]R) -- node[right,sa/font small] {$\Delta f$} ([xshift=3mm]Q);

  \node[sa/equation box,anchor=north west] at (0.35,3.02)
    {$m_{\mathrm{sec}}=\dfrac{\Delta f}{\Delta x}
      =\dfrac{f(x_0+\Delta x)-f(x_0)}{\Delta x}$};
\end{tikzpicture}
\end{document}
